ESDA

  • 9999.99s and 0s detected in nutrients (NAs) clean by impute by mean (Baker et al.)
sp_covs$ExMg <- na_if(sp_covs$ExMg, -9999.00) # replace -9999 with na
sp_covs$ExK <- na_if(sp_covs$ExK, -9999.00)
sp_covs$ExMg <- na_if(sp_covs$ExMg, 0.00) # replace 0 with na
sp_covs$ExK <- na_if(sp_covs$ExK, 0.00)
# replace per column
sp_covs$ExK[is.na(sp_covs$ExK)] <- mean(sp_covs$ExK, na.rm = T)
sp_covs$ExMg[is.na(sp_covs$ExMg)] <- mean(sp_covs$ExMg, na.rm = T)
